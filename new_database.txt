-- =============================
-- USERS & AUTHENTICATION
-- =============================
CREATE TABLE users (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(100) UNIQUE NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    role ENUM('admin','veterinarian','staff','reception','pharmacy','groomer','boarding','pet_owner') NOT NULL,
    first_name VARCHAR(100),
    last_name VARCHAR(100),
    contact_number VARCHAR(20),
    address TEXT,
    is_active TINYINT(1) DEFAULT 1,
    email_verified TINYINT(1) DEFAULT 0,
    phone_verified TINYINT(1) DEFAULT 0,
    last_login DATETIME,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_role (role),
    INDEX idx_email (email),
    INDEX idx_active (is_active)
) ENGINE=InnoDB;

-- =============================
-- PET OWNERS
-- =============================
CREATE TABLE pet_owners (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    notes TEXT,
    preferred_contact_method ENUM('email','sms') DEFAULT 'email',
    emergency_contact_name VARCHAR(150),
    emergency_contact_phone VARCHAR(20),
    emergency_contact_relationship VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user (user_id)
) ENGINE=InnoDB;

-- =============================
-- PETS (UPDATED - WITH REGISTRATION NUMBER)
-- =============================
CREATE TABLE pets (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    
    -- ✅ NEW: Auto-generated registration number (user-facing ID)
    registration_number VARCHAR(20) UNIQUE NOT NULL COMMENT 'User-facing unique identifier (e.g., 2026-000001)',
    
    owner_id BIGINT UNSIGNED NOT NULL,
    name VARCHAR(100) NOT NULL,
    species VARCHAR(50) NOT NULL,
    breed VARCHAR(100),
    birth_date DATE,
    gender ENUM('male','female','unknown') DEFAULT 'unknown',
    color VARCHAR(100),
    weight DECIMAL(5,2),
    microchip_number VARCHAR(50) UNIQUE,
    photo_path VARCHAR(255),
    qr_code_path VARCHAR(255),
    is_active TINYINT(1) DEFAULT 1,
    deceased_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    FOREIGN KEY (owner_id) REFERENCES pet_owners(id) ON DELETE CASCADE,
    
    INDEX idx_owner (owner_id),
    INDEX idx_species (species),
    INDEX idx_microchip (microchip_number),
    INDEX idx_active (is_active),
    INDEX idx_pet_registration (registration_number),
    
    CONSTRAINT chk_weight CHECK (weight IS NULL OR (weight > 0 AND weight <= 500))
) ENGINE=InnoDB;

-- =============================
-- APPOINTMENTS
-- =============================
CREATE TABLE appointments (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    pet_id BIGINT UNSIGNED NOT NULL,
    veterinarian_id BIGINT UNSIGNED,
    appointment_date DATETIME NOT NULL,
    type ENUM('consultation','vaccination','surgery','grooming','boarding','follow_up','emergency','other') DEFAULT 'consultation',

    FOREIGN KEY (pet_id) REFERENCES pets(id) ON DELETE CASCADE,
    FOREIGN KEY (veterinarian_id) REFERENCES users(id) ON DELETE SET NULL,
    INDEX idx_pet (pet_id),
    INDEX idx_vet (veterinarian_id),
    INDEX idx_date (appointment_date),
    INDEX idx_status (status),
    INDEX idx_queue (queue_status, queue_priority),
    INDEX idx_appointment_search (appointment_date, status, veterinarian_id)
) ENGINE=InnoDB;

-- =============================
-- INVOICING (UPDATED - WITH AUTO-GENERATION)
-- =============================
CREATE TABLE invoices (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    order_id BIGINT UNSIGNED,
    appointment_id BIGINT UNSIGNED,
    pet_id BIGINT UNSIGNED,
    owner_id BIGINT UNSIGNED NOT NULL,
    
    -- ✅ NEW: Invoice auto-generation fields
    invoice_prefix VARCHAR(10) DEFAULT 'INV' COMMENT 'Invoice prefix (e.g., INV, NYC-INV for multi-location)',
    invoice_sequence INT UNSIGNED COMMENT 'Sequential number within year/prefix',
    
    invoice_number VARCHAR(50) UNIQUE NOT NULL COMMENT 'Auto-generated: [prefix]-YYYY-NNNNNN (e.g., INV-2026-000001)',
    issue_date DATE NOT NULL,
    due_date DATE NOT NULL,
    tax_rate DECIMAL(5,2) DEFAULT 0.00 COMMENT 'Tax percentage to apply',
    discount_amount DECIMAL(12,2) DEFAULT 0.00 COMMENT 'Manual discount amount',
    discount_reason VARCHAR(255) COMMENT 'Reason for discount',
    status ENUM('pending','partial','paid','overdue','cancelled') DEFAULT 'pending',
    notes TEXT,
    overdue_reminder_sent TINYINT(1) DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE SET NULL,
    FOREIGN KEY (appointment_id) REFERENCES appointments(id) ON DELETE SET NULL,
    FOREIGN KEY (pet_id) REFERENCES pets(id) ON DELETE SET NULL,
    FOREIGN KEY (owner_id) REFERENCES pet_owners(id) ON DELETE CASCADE,
    INDEX idx_owner (owner_id),
    INDEX idx_invoice_number (invoice_number),
    INDEX idx_status (status),
    INDEX idx_dates (issue_date, due_date),
    INDEX idx_invoice_search (owner_id, status, due_date),
    INDEX idx_invoice_generation (invoice_prefix, invoice_sequence),
    CONSTRAINT chk_invoice_amounts CHECK (
